/* ── Paste this ENTIRE code into Apps Script, then redeploy ── */ function doGet(e) { var p = e.parameter || {}; /* GPS — kira jarak jalan raya antara 2 alamat */ if (p.action === 'distance') { try { var dir = Maps.newDirectionFinder() .setOrigin(p.from).setDestination(p.to).setMode(Maps.DirectionFinder.Mode.DRIVING) .getDirections(); if (!dir.routes || !dir.routes.length) return out(JSON.stringify({error:'Alamat tidak dijumpai'})); var leg = dir.routes[0].legs[0]; return out(JSON.stringify({ km: (leg.distance.value/1000).toFixed(2), mins: Math.round(leg.duration.value/60) })); } catch(err) { return out(JSON.stringify({error:String(err)})); } } /* GPS — tukar koordinat kepada alamat */ if (p.action === 'geocode') { try { var r = Maps.newGeocoder().reverseGeocode(Number(p.lat), Number(p.lng)); if (r.results && r.results.length) return out(JSON.stringify({address: r.results[0].formatted_address})); return out(JSON.stringify({error:'Alamat tidak dijumpai'})); } catch(err) { return out(JSON.stringify({error:String(err)})); } } /* Tarik senarai tugasan */ if (p.action === 'getTasks') { return out(JSON.stringify( getSheet('Tugasan').getDataRange().getValues() )); } return out(JSON.stringify( getSheet('Log').getDataRange().getValues() )); } function doPost(e) { const d = JSON.parse(e.postData.contents); if (d.action === 'add') { getSheet('Log').appendRow([d.id,d.datetime,d.title,d.category, d.duration,d.tags,d.notes,d.attachUrl,d.attachName, d.claimJson||'',d.otMula||'',d.otTamat||'',d.otDur||'']); } else if (d.action === 'delete') { const sh=getSheet('Log'), r=sh.getDataRange().getValues(); for (let i=r.length-1;i>=1;i--) if (r[i][0]==d.id) { sh.deleteRow(i+1); break; } } else if (d.action === 'update') { const sh=getSheet('Log'), r=sh.getDataRange().getValues(); for (let i=r.length-1;i>=1;i--) if (r[i][0]==d.id) { sh.getRange(i+1,1,1,13).setValues([[d.id,d.datetime,d.title, d.category,d.duration,d.tags,d.notes, d.attachUrl,d.attachName,d.claimJson||'', d.otMula||'',d.otTamat||'',d.otDur||'']]); break; } } else if (d.action === 'addClaim') { getSheet('Tuntutan').appendRow([d.logId,d.tarikh, d.tujuanP||d.tujuan||'',d.dari,d.ke,d.jarak, d.bertolakP,d.tibaP,d.tollP, d.tujuanB||'',d.dariB||'',d.keB||'',d.jarakB||'', d.bertolakB||'',d.tibaB||'',d.tollB||'', d.parking,d.hotel,d.pelbagai,d.jumlah,new Date()]); } else if (d.action === 'saveTask') { var sh = getSheet('Tugasan'); var r = sh.getDataRange().getValues(); var row = [d.id,d.title,d.done,d.prio,d.due,d.cat, d.note,d.subtasks,d.created,d.doneAt]; var found = false; for (var i=1;i<r.length;i++) if (String(r[i][0])==String(d.id)) { sh.getRange(i+1,1,1,10).setValues([row]); found=true; break; } if (!found) sh.appendRow(row); } else if (d.action === 'delTask') { var sh2 = getSheet('Tugasan'); var r2 = sh2.getDataRange().getValues(); for (var j=r2.length-1;j>=1;j--) if (String(r2[j][0])==String(d.id)) { sh2.deleteRow(j+1); break; } } else if (d.action === 'upload') { const fold = DriveApp.getFoldersByName('PKPJ Logbook').hasNext() ?
